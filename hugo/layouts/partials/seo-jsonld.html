{{- /* Structured data for the blog. Home: Organization + WebSite (SearchAction). Every other page: BreadcrumbList. Home › Section › Page (project notes add the Projects level). */ -}} {{- $crumbs := slice (dict "@type" "ListItem" "position" 1 "name" site.Title "item" site.Home.Permalink) -}} {{- $position := 2 -}} "name" (.Title | plainify) "item" .Permalink) -}} {{- if .IsPage -}} {{- $seoDesc := partialCached "seo-metadata-description.html" . .RelPermalink -}} {{- $type := "Article" -}} {{- if eq .Section "posts" }}{{ $type = "BlogPosting" }}{{ end -}} {{- if .Params.repository_note }}{{ $type = "TechArticle" }}{{ end -}} {{- $node := dict "@context" "https://schema.org" "@type" $type "@id" (printf "%s#article" .Permalink) "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "headline" (.Title | plainify) "url" .Permalink "inLanguage" .Lang "isAccessibleForFree" true "author" $person "2006-01-02T15:04:05-07:00")) }}{{ end -}} {{- with .Lastmod }}{{ $node = merge $node (dict "dateModified" (.Format "2006-01-02T15:04:05-07:00")) }}{{ end -}} {{- with .GetTerms "tags" -}} {{- $keywords := slice -}} {{- range . }}{{ $keywords = $keywords | append (.Title | plainify) }}{{ end -}} {{- $node = merge $node (dict "keywords" (delimit $keywords ", ")) -}} {{- end -}} {{- with .WordCount }}{{ $node = merge $node (dict "wordCount" .) }}{{ end -}} {{- $images := slice -}} {{- with partial "_funcs/get-page-images" . -}} {{- range first 3 . }}{{ $images = $images | append .Permalink }}{{ end -}} {{- end -}} {{- if gt (len $images) 0 }}{{ $node = merge $node (dict "image" $images) }}{{ end -}} {{- /* Translations of the same article. */ -}} {{- if .IsTranslated -}} {{- $works := slice -}} {{- range .Translations }}{{ $works = $works | append (dict "@type" $type "@id" (printf "%s#article" .Permalink) "url" .Permalink "inLanguage" .Lang) }}{{ end -}} {{- $node = merge $node (dict "workTranslation" $works) -}} {{- end -}} {{- /* Project notes describe one public repository. */ -}} {{- with .Params.repository_note -}} {{- $repo := . -}} {{- $note := index (where $note }}{{ with .url }}{{ $codeUrl = . }}{{ end }}{{ end -}} {{- $code := dict "@type" "SoftwareSourceCode" "name" $repo "codeRepository" $codeUrl "url" $codeUrl "author" $person -}} {{- with (dict "about" $code) -}} {{- end -}} {{- if eq .Section "recipes" -}} {{- $node = merge $node (dict "@type" "Article" "articleSection" "Recipes") -}} {{- end -}} {{- end -}} {{- end -}}
